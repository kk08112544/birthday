<script setup lang="ts">
import { ref, onMounted } from 'vue';
import { api } from 'src/boot/axios';
import { useQuasar } from 'quasar';
import type { AxiosError } from 'axios';

const $q = useQuasar();

// ─── Types ────────────────────────────────────────────────────────────────────
interface LogData {
  uId?: number;
  firstName?: string;
  userName?: string;
  role?: string;
  phoneNumber?: string;
  email?: string;
  createdAt?: string;
  updatedAt?: string;
  deletedAt?: string | null;
  [key: string]: unknown;
}

interface LogEntry {
  raw: string;
  timestamp: string;
  action: string;
  data?: LogData;
  oldData?: LogData;
  newData?: LogData;
}

// ─── State ────────────────────────────────────────────────────────────────────
const rows = ref<LogEntry[]>([]);
const loading = ref(false);
const search = ref('');
const activeAction = ref('ALL');
const page = ref(1);
const limit = ref(10);
const totalCount = ref(0);
const totalPages = ref(1);
const createCount = ref(0);
const updateCount = ref(0);
const deleteCount = ref(0);

const detailDialog = ref(false);
const selectedEntry = ref<LogEntry | null>(null);

// ─── Action filter options ─────────────────────────────────────────────────
const actionOptions = [
  { value: 'ALL', label: 'ทั้งหมด', icon: 'list' },
  { value: 'CREATE', label: 'สร้าง', icon: 'add_circle_outline' },
  { value: 'UPDATE', label: 'แก้ไข', icon: 'edit_note' },
  { value: 'DELETE', label: 'ลบ', icon: 'delete_outline' },
];

// ─── Helpers ──────────────────────────────────────────────────────────────────
const getInitial = (name: string): string => {
  if (!name) return '?';
  const parts = name.split('-');
  const target = parts.length > 1 ? parts[parts.length - 1] : parts[0];
  return (target?.trim().charAt(0) ?? '?').toUpperCase();
};

const getRoleLabel = (role?: string): string => {
  const map: Record<string, string> = { superAdmin: 'Super Admin', admin: 'Admin' };
  return role ? (map[role] ?? role) : '-';
};

const getRoleClass = (role?: string): string => {
  const map: Record<string, string> = {
    superAdmin: 'role-badge-sm--super',
    admin: 'role-badge-sm--admin',
  };
  return role ? (map[role] ?? 'role-badge-sm--admin') : 'role-badge-sm--admin';
};

const getActionIcon = (action: string | undefined): string => {
  const map: Record<string, string> = {
    CREATE: 'add_circle_outline',
    UPDATE: 'edit_note',
    DELETE: 'delete_outline',
  };
  return (action && map[action]) ? map[action]! : 'info_outline';
};

const getHeaderClass = (action: string | undefined): string => {
  const map: Record<string, string> = {
    CREATE: 'dialog-header--create',
    UPDATE: 'dialog-header--update',
    DELETE: 'dialog-header--danger',
  };
  return (action && map[action]) ? map[action]! : 'dialog-header--indigo';
};

const getHeaderIconClass = (action: string | undefined): string => {
  const map: Record<string, string> = {
    CREATE: 'dialog-header-icon--create',
    UPDATE: 'dialog-header-icon--update',
    DELETE: 'dialog-header-icon--danger',
  };
  return (action && map[action]) ? map[action]! : 'dialog-header-icon--indigo';
};

const formatDate = (ts: string | undefined): string => {
  if (!ts) return '-';
  return new Intl.DateTimeFormat('th-TH', {
    year: 'numeric',
    month: 'short',
    day: 'numeric',
    hour: '2-digit',
    minute: '2-digit',
    second: '2-digit',
  }).format(new Date(ts));
};

const getPreview = (entry: LogEntry): string => {
  if (entry.action === 'UPDATE') {
    const name = entry.newData?.firstName ?? entry.oldData?.firstName ?? '';
    const user = entry.newData?.userName ?? entry.oldData?.userName ?? '';
    return `อัปเดตข้อมูล: ${name} (@${user})`;
  }
  const d = entry.data;
  if (!d) return entry.raw.slice(0, 80);
  return `${d.firstName ?? ''} (@${d.userName ?? ''}) — ${d.email ?? ''}`;
};

// ─── Parse raw log line ───────────────────────────────────────────────────────
const parseLogLine = (line: string): LogEntry => {
  // Format: "2026-05-20T21:12:03.927Z [CREATE] {...}"
  // or UPDATE: "timestamp [UPDATE] OLD: {...} | NEW: {...}"
  const tsMatch = line.match(/^(\S+Z)/);
  const timestamp: string = tsMatch?.[1] ?? '';
  const actionMatch = line.match(/\[(CREATE|UPDATE|DELETE)\]/);
  const action: string = actionMatch?.[1] ?? 'UNKNOWN';

  if (action === 'UPDATE') {
    const oldMatch = line.match(/OLD:\s*(\{.*?\})\s*\|/s);
    const newMatch = line.match(/\|\s*NEW:\s*(\{.*\})/s);
    try {
      const oldData: LogData = oldMatch ? JSON.parse(oldMatch[1] ?? '{}') : {};
      const newData: LogData = newMatch ? JSON.parse(newMatch[1] ?? '{}') : {};
      return { raw: line, timestamp, action, oldData, newData };
    } catch {
      return { raw: line, timestamp, action };
    }
  } else {
    const jsonMatch = line.match(/\]\s*(\{.*\})/s);
    try {
      const data: LogData = jsonMatch ? JSON.parse(jsonMatch[1] ?? '{}') : {};
      return { raw: line, timestamp, action, data };
    } catch {
      return { raw: line, timestamp, action };
    }
  }
};

// ─── Fetch ────────────────────────────────────────────────────────────────────
const fetchLogs = async () => {
  loading.value = true;
  try {
    const response = await api.get('/backoffice/admin/log', {
      params: {
        page: page.value,
        limit: limit.value,
        search: search.value.trim() || undefined,
        action: activeAction.value !== 'ALL' ? activeAction.value : undefined,
      },
    });
    const res = response.data;
    totalCount.value = res.total ?? 0;
    totalPages.value = res.totalPages ?? 1;
    createCount.value = res.createCount ?? 0;
    updateCount.value = res.updateCount ?? 0;
    deleteCount.value = res.deleteCount ?? 0;

    rows.value = (res.data as string[]).map(parseLogLine);
  } catch (err: unknown) {
    const error = err as AxiosError<{ message: string }>;
    console.error('fetch log error', error.response?.data?.message);
    rows.value = [];
  } finally {
    loading.value = false;
  }
};

// ─── Controls ─────────────────────────────────────────────────────────────────
const setAction = (val: string) => {
  activeAction.value = val;
  page.value = 1;
  void fetchLogs();
};

const onSearch = () => {
  page.value = 1;
  void fetchLogs();
};

const clearSearch = () => {
  search.value = '';
  onSearch();
};

const prevPage = () => {
  if (page.value > 1) { page.value--; void fetchLogs(); }
};

const nextPage = () => {
  if (page.value < totalPages.value) { page.value++; void fetchLogs(); }
};

const onLimitChange = () => {
  page.value = 1;
  void fetchLogs();
};

// ─── Detail Dialog ────────────────────────────────────────────────────────────
const openDetail = (entry: LogEntry) => {
  selectedEntry.value = entry;
  detailDialog.value = true;
};

// ─── Lifecycle ────────────────────────────────────────────────────────────────
onMounted(() => void fetchLogs());
</script>
