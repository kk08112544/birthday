<script setup lang="ts">
import { ref, computed, onMounted, onUnmounted } from 'vue';
import { api } from 'src/boot/axios';
import { useQuasar } from 'quasar';
import type { QTableProps, QTableColumn } from 'quasar';

// ─── Types ────────────────────────────────────────────────────────────────────
interface LogUser { uId: number; firstName: string; userName: string; }
interface LogData {
  upId: number | string; word: string;
  createdByUser?: LogUser | null; updatedByUser?: LogUser | null; deletedByUser?: LogUser | null;
  [key: string]: unknown;
}
interface LogEntry {
  ts: string; action: 'CREATE' | 'UPDATE' | 'DELETE';
  data: LogData | null; oldData: LogData | null; newData: LogData | null;
}
interface TableRow {
  rowKey: string; displayIndex: number;
  action: 'CREATE' | 'UPDATE' | 'DELETE';
  word: string; diffs: { key: string; old: string; new: string }[];
  actor: string; ts: string; upId: number | string;
}
interface Particle { id: number; style: Record<string, string>; }
type ShapeType = 'circle' | 'square' | 'star' | 'triangle' | 'emoji';
type FilterType = 'all' | 'CREATE' | 'UPDATE' | 'DELETE';

const $q = useQuasar();
const rows = ref<TableRow[]>([]);
const loading = ref(false);
const search = ref('');
const activeFilter = ref<FilterType>('all');
const activeParticles = ref<Particle[]>([]);
let particleId = 0;

const stats = ref({ total: 0, createCount: 0, updateCount: 0, deleteCount: 0 });
const serverPage = ref(1);
const serverLimit = ref(10);
const serverTotal = ref(0);

const tablePagination = ref({
  page: 1, rowsPerPage: 10, rowsNumber: 0, sortBy: '', descending: false,
});

const columns: QTableColumn[] = [
  { name: 'no',     label: 'ลำดับ',           field: 'displayIndex', align: 'center', style: 'width:60px' },
  { name: 'action', label: 'ประเภท',           field: 'action',       align: 'left',   style: 'width:110px' },
  { name: 'word',   label: 'คำ',               field: 'word',         align: 'left',   style: 'width:160px' },
  { name: 'diff',   label: 'การเปลี่ยนแปลง',  field: 'diffs',        align: 'left' },
  { name: 'actor',  label: 'ผู้ดำเนินการ',     field: 'actor',        align: 'left',   style: 'width:210px' },
  { name: 'ts',     label: 'วันที่/เวลา',      field: 'ts',           align: 'left',   style: 'width:140px' },
];

const filterOptions = computed(() => [
  { value: 'all' as const,    label: 'ทั้งหมด', icon: 'list_alt',          count: stats.value.createCount + stats.value.updateCount + stats.value.deleteCount },
  { value: 'CREATE' as const, label: 'สร้าง',   icon: 'add_circle_outline', count: stats.value.createCount },
  { value: 'UPDATE' as const, label: 'แก้ไข',   icon: 'edit_note',          count: stats.value.updateCount },
  { value: 'DELETE' as const, label: 'ลบ',       icon: 'delete_outline',     count: stats.value.deleteCount },
]);

function parseLogLine(line: string): LogEntry | null {
  const tsMatch = line.match(/^(\S+Z)/);
  const actionMatch = line.match(/\[(CREATE|UPDATE|DELETE)\]/);
  if (!tsMatch || !actionMatch) return null;
  const ts = tsMatch[1]!;
  const action = actionMatch[1]! as 'CREATE' | 'UPDATE' | 'DELETE';
  let data: LogData | null = null, oldData: LogData | null = null, newData: LogData | null = null;
  if (action === 'UPDATE') {
    const m = line.match(/OLD: (\{.*?\}) \| NEW: (\{.*\})$/);
    if (m) { try { oldData = JSON.parse(m[1]!); } catch (_) {} try { newData = JSON.parse(m[2]!); } catch (_) {} }
  } else {
    const m = line.match(/\[(?:CREATE|DELETE)\] (\{.*\})$/);
    if (m) { try { data = JSON.parse(m[1]!); } catch (_) {} }
  }
  return { ts, action, data, oldData, newData };
}

function getWord(e: LogEntry): string {
  if (e.action === 'CREATE' && e.data) return String(e.data.word ?? '-');
  if (e.action === 'UPDATE' && e.newData) return String(e.newData.word ?? '-');
  if (e.action === 'DELETE' && e.data) return String(e.data.word ?? '-');
  return '-';
}
function getActor(e: LogEntry): string {
  const user = e.action === 'CREATE' ? e.data?.createdByUser
    : e.action === 'UPDATE' ? e.newData?.updatedByUser
    : e.data?.deletedByUser;
  return user ? `${user.firstName} (${user.userName})` : '-';
}
function getUpId(e: LogEntry): number | string { return (e.data ?? e.newData ?? e.oldData)?.upId ?? '-'; }
function getDiffs(e: LogEntry) {
  if (!e.oldData || !e.newData) return [];
  return ['word'].filter(k => String(e.oldData![k]) !== String(e.newData![k]))
    .map(k => ({ key: k, old: String(e.oldData![k]), new: String(e.newData![k]) }));
}

function getInitial(actor: string): string {
  if (actor === '-') return '?';
  return actor.charAt(0).toUpperCase();
}
function getActorName(actor: string): string {
  return actor.includes('(') ? actor.split('(')[0]!.trim() : actor;
}
function getActorUsername(actor: string): string {
  const m = actor.match(/\(([^)]+)\)/);
  return m ? `@${m[1]}` : '';
}
function getActorUsernameRaw(actor: string): string {
  const m = actor.match(/\(([^)]+)\)/);
  return m ? m[1]! : '';
}

function entryToRow(entry: LogEntry, index: number): TableRow {
  return {
    rowKey: `${entry.ts}-${index}`,
    displayIndex: (serverPage.value - 1) * serverLimit.value + index + 1,
    action: entry.action,
    word: getWord(entry),
    diffs: getDiffs(entry),
    actor: getActor(entry),
    ts: entry.ts,
    upId: getUpId(entry),
  };
}

const fetchLog = async (): Promise<void> => {
  loading.value = true;
  try {
    const res = await api.get('/backoffice/unpolite/log', {
      params: {
        page: serverPage.value, limit: serverLimit.value,
        search: search.value.trim() || undefined,
        action: activeFilter.value === 'all' ? undefined : activeFilter.value,
      },
    });
    const d = res.data;
    const entries: LogEntry[] = (d.data as string[]).map(parseLogLine).filter((e): e is LogEntry => e !== null);
    rows.value = entries.map((entry, i) => entryToRow(entry, i));
    serverTotal.value = d.total;
    tablePagination.value.rowsNumber = d.total;
    tablePagination.value.page = serverPage.value;
    tablePagination.value.rowsPerPage = serverLimit.value;
    stats.value = { total: d.total, createCount: d.createCount, updateCount: d.updateCount, deleteCount: d.deleteCount };
  } catch { rows.value = []; } finally { loading.value = false; }
};

const onTableRequest: QTableProps['onRequest'] = (reqProps) => {
  serverPage.value = reqProps.pagination.page;
  serverLimit.value = reqProps.pagination.rowsPerPage;
  tablePagination.value.page = reqProps.pagination.page;
  tablePagination.value.rowsPerPage = reqProps.pagination.rowsPerPage;
  void fetchLog();
};
function onSearch() { serverPage.value = 1; tablePagination.value.page = 1; void fetchLog(); }
function onRefresh() { serverPage.value = 1; tablePagination.value.page = 1; void fetchLog(); }
function setFilter(f: FilterType) { activeFilter.value = f; serverPage.value = 1; tablePagination.value.page = 1; void fetchLog(); }
function actionIcon(a: string) { return a === 'CREATE' ? 'add_circle_outline' : a === 'UPDATE' ? 'edit_note' : 'delete_outline'; }
function actionLabel(a: string) { return a === 'CREATE' ? 'สร้าง' : a === 'UPDATE' ? 'แก้ไข' : 'ลบ'; }
function formatDateOnly(ts: string) { return new Date(ts).toLocaleDateString('th-TH', { day: '2-digit', month: 'short', year: 'numeric' }); }
function formatTimeOnly(ts: string) { return new Date(ts).toLocaleTimeString('th-TH', { hour: '2-digit', minute: '2-digit', second: '2-digit' }); }

// ─── Particles ────────────────────────────────────────────────────────────────
const PARTICLE_COLORS = ['#e11d48','#fbbf24','#6366f1','#22c55e','#fb7185','#f59e0b','#a78bfa','#34d399','#f472b6','#38bdf8'];
const PARTICLE_EMOJIS = ['🎉','✨','🎊','⭐','💫','🌟','🎈','🌸','🌺','🌼'];
const spawnParticles = (x: number, y: number) => {
  const count = 10 + Math.floor(Math.random() * 5);
  for (let i = 0; i < count; i++) {
    const id = ++particleId;
    const size = 5 + Math.random() * 6;
    const color = PARTICLE_COLORS[Math.floor(Math.random() * PARTICLE_COLORS.length)]!;
    const dur = 0.6 + Math.random() * 0.5;
    const emoji = PARTICLE_EMOJIS[Math.floor(Math.random() * PARTICLE_EMOJIS.length)]!;
    const shapes: ShapeType[] = ['circle','circle','square','star','emoji'];
    const shape = shapes[Math.floor(Math.random() * shapes.length)]!;
    const isEmoji = shape === 'emoji';
    const angle = (i / count) * Math.PI * 2 + (Math.random() - 0.5) * 0.8;
    const dist = 70 + Math.random() * 70;
    const style: Record<string, string> = {
      '--x': `${x}px`, '--y': `${y}px`,
      '--dx': `${Math.cos(angle) * dist}px`, '--dy': `${Math.sin(angle) * dist}px`,
      '--dur': `${dur}s`, '--color': isEmoji ? 'transparent' : color,
      '--size': isEmoji ? '16px' : `${size}px`, '--shape': shape,
      '--emoji-content': isEmoji ? `"${emoji}"` : '""',
      '--rot': `${Math.random() * 360 - 180}deg`,
    };
    activeParticles.value.push({ id, style });
    setTimeout(() => { activeParticles.value = activeParticles.value.filter(p => p.id !== id); }, dur * 1000 + 100);
  }
};
const handleGlobalClick = (e: MouseEvent) => spawnParticles(e.clientX, e.clientY);
onMounted(() => { void fetchLog(); document.addEventListener('click', handleGlobalClick); });
onUnmounted(() => { document.removeEventListener('click', handleGlobalClick); });
</script>
